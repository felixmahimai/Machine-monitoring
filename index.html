Set up the collector PC

You or your developer
# firewall and no routing
echo 'net.ipv4.ip_forward=0' | sudo tee /etc/sysctl.d/99-noforward.conf
sudo sysctl --system
sudo ufw default deny incoming
sudo ufw allow in on enp1s0 to any port 1883 proto tcp
sudo ufw allow in on enp1s0 to any port 3000 proto tcp
sudo ufw enable
sudo apt install chrony docker.io docker-compose-v2 unattended-upgrades
